import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return new Response("Method not allowed", { status: 405, headers: corsHeaders }); } const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SERVICE_ROLE_KEY = Deno.env.get("SERVICE_ROLE_KEY")!; const SM8_APP_ID = Deno.env.get("SM8_APP_ID")!; const SM8_APP_SECRET = Deno.env.get("SM8_APP_SECRET")!; function sbHeaders() { return { "apikey": SERVICE_ROLE_KEY, "Authorization": `Bearer ${SERVICE_ROLE_KEY}`, "Content-Type": "application/json", }; } try { const body = await req.json(); const { jobNumber, lines } = body; // lines: array of strings, e.g. ["📦 12 x Colorbond Corrugated — Surfmist (by Dave)"] if (!jobNumber || !Array.isArray(lines) || lines.length === 0) { return new Response(JSON.stringify({ error: "Missing jobNumber or lines[]" }), { status: 400, headers: corsHeaders }); } // 1. Get ServiceM8 tokens const tokenRowRes = await fetch(`${SUPABASE_URL}/rest/v1/sm8_tokens?id=eq.1&select=*`, { headers: { "apikey": SERVICE_ROLE_KEY, "Authorization": `Bearer ${SERVICE_ROLE_KEY}` }, }); const tokenRows = await tokenRowRes.json(); if (!tokenRows.length) { return new Response(JSON.stringify({ error: "No ServiceM8 connection found. Please reconnect." }), { status: 500, headers: corsHeaders }); } let { access_token, refresh_token, expires_at } = tokenRows[0]; // 2. Refresh access token if expired if (new Date(expires_at) <= new Date()) { const refreshRes = await fetch("https://go.servicem8.com/oauth/access_token", { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: new URLSearchParams({ grant_type: "refresh_token", client_id: SM8_APP_ID, client_secret: SM8_APP_SECRET, refresh_token: refresh_token, }), }); const refreshData = await refreshRes.json(); if (!refreshRes.ok) { return new Response(JSON.stringify({ error: "Token refresh failed", details: refreshData }), { status: 500, headers: corsHeaders }); } access_token = refreshData.access_token; refresh_token = refreshData.refresh_token || refresh_token; await fetch(`${SUPABASE_URL}/rest/v1/sm8_tokens`, { method: "POST", headers: { ...sbHeaders(), "Prefer": "resolution=merge-duplicates" }, body: JSON.stringify({ id: 1, access_token, refresh_token, expires_at: new Date(Date.now() + (refreshData.expires_in || 3600) * 1000).toISOString(), }), }); } // 3. Look up job by generated_job_id const jobLookupRes = await fetch( `https://api.servicem8.com/api_1.0/job.json?$filter=generated_job_id eq '${jobNumber}'`, { headers: { Authorization: `Bearer ${access_token}` } } ); const jobs = await jobLookupRes.json(); if (!jobLookupRes.ok) { return new Response(JSON.stringify({ error: "Job lookup failed", details: jobs }), { status: 500, headers: corsHeaders }); } if (!Array.isArray(jobs) || jobs.length === 0) { return new Response(JSON.stringify({ error: `No ServiceM8 job found matching "${jobNumber}"` }), { status: 404, headers: corsHeaders }); } const jobUuid = jobs[0].uuid; // 4. Check if we already have a running note for this job const existingRes = await fetch( `${SUPABASE_URL}/rest/v1/sm8_allocation_notes?job_number=eq.${encodeURIComponent(jobNumber)}&select=*`, { headers: { "apikey": SERVICE_ROLE_KEY, "Authorization": `Bearer ${SERVICE_ROLE_KEY}` } } ); const existingRows = await existingRes.json(); const existing = existingRows && existingRows[0]; const newLineBlock = lines.map((l: string) => "• " + l).join("\n"); const today = new Date().toLocaleDateString("en-AU", { day: "2-digit", month: "short", year: "numeric" }); if (existing && existing.note_uuid) { // Try to append to the existing note const updatedBody = existing.note_body + "\n" + newLineBlock; const updateRes = await fetch(`https://api.servicem8.com/api_1.0/note/${existing.note_uuid}.json`, { method: "POST", headers: { Authorization: `Bearer ${access_token}`, "Content-Type": "application/json" }, body: JSON.stringify({ note: updatedBody }), }); if (updateRes.ok) { await fetch(`${SUPABASE_URL}/rest/v1/sm8_allocation_notes?job_number=eq.${encodeURIComponent(jobNumber)}`, { method: "PATCH", headers: sbHeaders(), body: JSON.stringify({ note_body: updatedBody }), }); return new Response(JSON.stringify({ success: true, mode: "updated" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // If update failed (e.g. note was deleted in ServiceM8), fall through and create a new one } // 5. Create a new note const header = "📦 Stock Allocation — " + jobNumber + " (" + today + ")"; const fullBody = header + "\n" + newLineBlock; const noteRes = await fetch("https://api.servicem8.com/api_1.0/note.json", { method: "POST", headers: { Authorization: `Bearer ${access_token}`, "Content-Type": "application/json" }, body: JSON.stringify({ related_object: "job", related_object_uuid: jobUuid, note: fullBody }), }); if (!noteRes.ok) { const errText = await noteRes.text(); return new Response(JSON.stringify({ error: "Note creation failed", details: errText }), { status: 500, headers: corsHeaders }); } const newNoteUuid = noteRes.headers.get("x-record-uuid"); await fetch(`${SUPABASE_URL}/rest/v1/sm8_allocation_notes`, { method: "POST", headers: { ...sbHeaders(), "Prefer": "resolution=merge-duplicates" }, body: JSON.stringify({ job_number: jobNumber, note_uuid: newNoteUuid, note_body: fullBody }), }); return new Response(JSON.stringify({ success: true, mode: "created" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } catch (err) { return new Response(JSON.stringify({ error: String(err) }), { status: 500, headers: corsHeaders }); } });
